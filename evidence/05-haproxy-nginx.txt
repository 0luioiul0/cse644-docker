#---------------------------------------------------------------------
# CSE644 requirement 7: HAProxy acting as a proxy in front of Nginx.
# Two Nginx backends are declared so the evidence also shows load
# balancing and health checking, not just a single forwarded request.
#---------------------------------------------------------------------

global
    log stdout format raw local0 info
    maxconn 2048

defaults
    log     global
    mode    http
    option  httplog
    option  dontlognull
    option  forwardfor          # pass the real client IP to Nginx
    timeout connect 5s
    timeout client  30s
    timeout server  30s
    retries 3
    option  redispatch          # retry on a DIFFERENT server, not the dead one

#---------------------------------------------------------------------
# Public entry point. Everything the client sees comes through here.
#---------------------------------------------------------------------
frontend http_in
    bind *:80

    # Prove at the client that the response passed through HAProxy
    http-response set-header X-Proxied-By "haproxy/%[env(HOSTNAME)]"
    http-request set-header X-Forwarded-Proto http

    default_backend nginx_pool

#---------------------------------------------------------------------
# Nginx pool. `check` uses the /health endpoint the Nginx image exposes.
#---------------------------------------------------------------------
backend nginx_pool
    balance roundrobin
    option httpchk GET /health
    http-check expect status 200

    # Docker's embedded DNS only exists once the containers are on the network,
    # so do not require web1/web2 to resolve while the config is being parsed.
    default-server init-addr last,libc,none

    # Tell the client which backend served the request
    http-response set-header X-Backend-Server %[srv_name]

    server web1 web1:80 check inter 3s fall 3 rise 2
    server web2 web2:80 check inter 3s fall 3 rise 2

#---------------------------------------------------------------------
# Stats page: visual proof that HAProxy sees both Nginx servers as UP.
#---------------------------------------------------------------------
listen stats
    bind *:8404
    stats enable
    stats uri /
    stats refresh 5s
    stats show-legends

$ cat /mnt/c/Users/zhang/OneDrive/Desktop/cse644-docker/03-haproxy-nginx/nginx/default.conf
# Log the client IP HAProxy forwarded, not HAProxy's own address.
# log_format belongs to the http context, so it lives outside the server block.
log_format proxied '$remote_addr fwd=$http_x_forwarded_for "$request" $status served_by=$hostname';

server {
    listen       80;
    server_name  _;

    root  /usr/share/nginx/html;
    index index.html;

    # $hostname is the container name, so the client can tell web1 from web2
    add_header X-Served-By $hostname always;

    access_log /var/log/nginx/access.log proxied;

    location / {
        # Rewrite the placeholder in the page with this container's hostname
        sub_filter '__BACKEND__' '$hostname';
        sub_filter_once off;
        sub_filter_types text/html;
        try_files $uri $uri/ =404;
    }

    location /whoami {
        default_type text/plain;
        return 200 "$hostname\n";
    }

    location /health {
        access_log off;
        default_type text/plain;
        return 200 "ok\n";
    }
}

$ cat /mnt/c/Users/zhang/OneDrive/Desktop/cse644-docker/03-haproxy-nginx/docker-compose.yml
# Requirement 7: HAProxy as a proxy in front of Nginx.
#
#   client -> localhost:8082 -> haproxy:80 -> (web1:80 | web2:80)
#
# Only HAProxy publishes host ports. The Nginx containers sit on the `edge`
# network with no port mapping, so from outside this machine the proxy is the
# only way in — that is what makes this a proxy demonstration rather than two
# independently exposed web servers.

name: cse644-proxy

services:
  haproxy:
    build: ./haproxy
    image: cse644-haproxy:1.0
    container_name: cse644-haproxy
    ports:
      - "8082:80"     # public entry point
      - "8404:8404"   # HAProxy stats page
    networks:
      - edge
    depends_on:
      web1:
        condition: service_healthy
      web2:
        condition: service_healthy
    restart: unless-stopped

  web1:
    build: ./nginx
    image: cse644-proxy-nginx:1.0
    container_name: cse644-web1
    hostname: web1
    networks:
      - edge
    # deliberately no `ports:` — not reachable from the host
    restart: unless-stopped

  web2:
    build: ./nginx                  # same build context as web1
    image: cse644-proxy-nginx:1.0   # same image, second replica
    container_name: cse644-web2
    hostname: web2
    networks:
      - edge
    depends_on:
      - web1
    restart: unless-stopped

networks:
  edge:
    name: cse644-edge
    driver: bridge
